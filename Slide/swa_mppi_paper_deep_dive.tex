\section{Motivation \& Problem}
% =============================================================================

\begin{frame}{The cultural-alignment gap in LLMs}
  \begin{itemize}
    \item \textbf{Moral preferences are culturally contingent.}\\
          The Moral Machine experiment collected \textbf{40M+ judgments from 233 countries}, finding systematic cross-cultural variation: East-Asian respondents weight \emph{youth} more; Southern-hemisphere populations show distinct utilitarian calculi; Islamic societies prioritise different social roles. \cit{Awad et al., Nature 2018}
    \item \textbf{LLMs encode a single, predominantly Western moral stance}---inherited from WEIRD-skewed training data. \cit{Henrich et al. 2010; Santurkar et al. 2023}
    \item \textbf{MultiTP} (Jin et al. 2025) operationalises this into country-specific Average Marginal Component Effects (AMCE) over \textbf{6 moral dimensions, 107 languages}; finds that \emph{none of 19 contemporary LLMs} achieve broadly satisfactory alignment.
  \end{itemize}
  \begin{block}{Research question}
    \emph{Can a frozen LLM be nudged toward country-specific moral preferences at \textbf{inference time}, using only what cross-cultural surveys already tell us?}
  \end{block}
\end{frame}

\begin{frame}{Why existing approaches don't fit}
  \small
  \begin{tabular}{p{3.4cm} p{4.0cm} p{5.5cm}}
    \toprule
    \textbf{Approach} & \textbf{Representative work} & \textbf{Limitation for cross-cultural alignment} \\
    \midrule
    Fine-tuning / RLHF
      & Bai et al.\ 2022; Ouyang et al.\ 2022
      & Per-culture labelled data; irreversible weight change \\
    Ad-hoc prompting
      & Jin et al.\ 2025
      & Weak empirical grounding \\
    Activation steering
      & Arditi et al.\ 2025; Zou et al.\ 2023
      & White-box; per-model engineering \\
    Reward-guided decoding
      & ARGS (Khanov 2024); Mudgal 2024
      & Trained reward model unavailable cross-culturally \\
    Culture adapters
      & CultureManager 2024; CAReDiO 2024
      & Per-culture labels; scale poorly $>$100 countries \\
    \bottomrule
  \end{tabular}
  \vspace{0.4em}

  \footnotesize\emph{None of these are training-free, black-box, country-conditioned, label-free, and weight-preserving simultaneously.}
\end{frame}

\begin{frame}{Core insight: \emph{disagreement is signal}}
  \begin{itemize}
    \item Within-country moral views are \textbf{not monolithic}---age cohorts \& social strata exhibit systematically different priorities. \cit{Awad et al. 2018; Kleiman-Weiner et al. 2017}
    \item The \textbf{World Values Survey (Wave 7)}~\cit{Haerpfer et al. 2022} provides empirically calibrated within-country variation across 80+ nations.
    \item \textbf{Build WVS-grounded persona agents} per country. When personas \emph{disagree} on a dilemma, that scenario is \textbf{culturally contested} → exploit through stochastic optimal control. \cit{Williams et al. 2017 (MPPI)}
    \item Apply a \textbf{bounded, KL-regularised correction} in \textbf{logit space} (not generation sampling), with a \textbf{Prospect Theory} utility that captures asymmetric gain/loss sensitivity. \cit{Kahneman \& Tversky 1979}
  \end{itemize}
  \begin{alertblock}{Key principle}
    \textbf{Self-limiting}: when personas agree, the method does \emph{nothing}. The intervention only fires where it matters.
  \end{alertblock}
\end{frame}

\section{Problem Formulation}
% =============================================================================

\begin{frame}{Setting: binary moral choice in MultiTP}
  \begin{itemize}
    \item Scenario $\mathbf{x}$ presents a \textbf{forced binary choice} via decision tokens \texttt{A} ($a$) and \texttt{B} ($b$).
    \item Frozen LLM $f_\theta$: produces logits $\mathbf{z}=[z_a,z_b]$, gap $\Delta = z_b - z_a$.
    \item Spare probability: $p_\text{spare} = \sigma(\Delta / T_\text{dec})$.
    \item \textbf{Human target}: country AMCE vector $\mathbf{h}^{(c)}\!\in\!\mathbb{R}^6$ from 40M+ Moral Machine judgments.
    \item \textbf{Model AMCE} (per dimension $d$):
      \[
        \hat{m}_d^{(c)} \;=\; \frac{1}{|\mathcal{S}_d^{(c)}|}\sum_{\mathbf{x}\in\mathcal{S}_d^{(c)}}
        p_\text{spare}(\mathbf{x})
      \]
      Five binary dims: empirical mean of $p_\text{spare}$. Utilitarianism: OLS regression on count difference $|n_\text{pref}-n_\text{non-pref}|$.
  \end{itemize}
  \begin{block}{Goal}
    Find logit correction $\delta^\ast(\mathbf{x},c)$, \emph{without modifying} $\theta$, so that $\hat{\mathbf{m}}^{(c)}$ is closer to $\mathbf{h}^{(c)}$ under \textbf{Jensen--Shannon divergence} (primary metric).
  \end{block}
\end{frame}

\section{Method: SWA-MPPI Pipeline}
% =============================================================================

\begin{frame}{Four-stage pipeline overview}
  \begin{enumerate}
    \item \textbf{Persona construction} ($N=4$): three WVS age cohorts + one utilitarian agent.
    \item \textbf{Batched evaluation}: base + $N$ personas in \emph{a single} forward pass → per-agent decision gaps $\delta_i$.
    \item \textbf{Adaptive conflict detection}: if inter-agent variance $\nu \ge \tau^{(c)}$, fire MPPI; otherwise return consensus only.
    \item \textbf{PT-weighted MPPI in logit space} + \textbf{positional debiasing}.
  \end{enumerate}
  \vspace{0.3em}
  \begin{block}{Deployment cost}
    One additional batched forward pass; the MPPI loop runs on \textbf{cached scalars} and is \textbf{free at the wall-clock level}.
  \end{block}
\end{frame}

\begin{frame}{Stage 1 — Persona construction (WVS → natural language)}
  \footnotesize
  \textbf{8 value dimensions} extracted from WVS Wave~7 microdata \cit{Haerpfer et al. 2022}:
  \begin{itemize}
    \item Gender egalitarianism (Q58P, Q59P, Q60P); Religiosity (Q6P); Trust (Q43P)
    \item Moral permissiveness (Q50, Q52P, Q54P); Work centrality (Q5P)
    \item Family importance (Q1P); Personal autonomy (Q39P); Meritocracy (Q40P)
  \end{itemize}
  \vspace{0.3em}
  \textbf{Procedure}: per country, segment respondents into \emph{young (18--35)}, \emph{middle (36--55)}, \emph{older ($>$55)}; compute per-cohort dimension means; map to natural-language descriptors (e.g.\ religiosity ratio $>$0.85 → ``deeply religious'').
  \vspace{0.3em}

  \textbf{Persona prompt template:}
  \begin{quote}\itshape\scriptsize
    You are a [age-cohort] adult from [country]. The cultural values typical of your social
    group are: [WVS statistics]. When making moral judgments, you tend to [WVS-grounded description].
  \end{quote}
  Fourth persona: country-invariant \textbf{utilitarian} agent (maximises lives saved).
  \begin{alertblock}{Why this matters}
    Personas reflect \textbf{measured population structure}, not author stereotypes---mitigating toxicity risks of naive cultural prompting. \cit{Deshpande et al.\ 2023}
  \end{alertblock}
\end{frame}

\begin{frame}{Stage 2 — Batched forward pass \& gaps}
  \textbf{Single batched forward pass} (base + $N$ personas):
  \[
    \mathbf{z}_\text{base},\;\mathbf{z}_1,\ldots,\mathbf{z}_N
    \;=\; f_\theta\!\left(\mathbf{x};\;\varnothing,\mathbf{s}_1,\ldots,\mathbf{s}_N\right).
  \]
  Per-category logit temperatures $T_\text{cat}\in\{4.0, 3.5, 1.5\}$ scale logits before computing gaps:
  \[
    \delta_i = \frac{z_{i,b}-z_{i,a}}{T_\text{cat}}, \qquad
    \bar{\delta} = \frac{1}{N}\sum_i \delta_i \;\text{(consensus)},\qquad
    r_i = \delta_i - \delta_\text{base}\;\text{(reward).}
  \]
  \begin{itemize}
    \item $\delta_i$: persona $i$'s decision gap on this scenario.
    \item $\bar{\delta}$: collective consensus.
    \item $r_i$: how strongly persona $i$ \emph{pulls away from the culturally-neutral baseline}---a large $|r_i|$ indicates a moral direction the vanilla model does not exhibit.
  \end{itemize}
\end{frame}

\begin{frame}{Stage 3 — Adaptive conflict detection $\tau^{(c)}$}
  \[
    \tau^{(c)} \;=\; \operatorname{Percentile}_{65}\!
    \Bigl(\bigl\{\operatorname{Var}(\delta_1^{(s)},\ldots,\delta_N^{(s)})\bigr\}_{s=1}^{n_\text{cal}}\Bigr),
    \quad n_\text{cal}=50.
  \]
  \[
    \text{MPPI fires} \iff \operatorname{Var}(\delta_1,\ldots,\delta_N) \ge \tau^{(c)}.
  \]
  \begin{itemize}
    \item Calibrated on \textbf{50 unlabelled scenarios}---uses only inter-agent variance; \textbf{never observes human AMCEs}.
    \item Targets a \textbf{35\% MPPI trigger rate} (65th percentile).
    \item Main experiments: 20\% held-out for $\tau$, 80\% for evaluation.
  \end{itemize}
  \begin{block}{Robustness evidence (Appendix \S A)}
    Held-out vs.\ in-distribution calibration over 15 countries: $|\Delta\text{JSD}| = \mathbf{0.0006}$
    (\emph{below} the bootstrap CI of $\pm 0.004$); 12/15 countries $A\le B$. → $\tau$ does \emph{not} overfit.
  \end{block}
\end{frame}

\begin{frame}{Stage 4a — Prospect Theory utility (per agent)}
  Cooperative decomposition with $\lambda_\text{coop}=0.7$:
  \[
    U_i(\epsilon_k) \;=\; (1-\lambda_\text{coop})\,v(r_i\,\tilde\delta_k)
    \;+\; \lambda_\text{coop}\,v\!\bigl(\bar r_{-i}\,\tilde\delta_k\bigr),\quad
    \tilde\delta_k = \bar\delta + \epsilon_k.
  \]
  Classical Kahneman--Tversky value function \cit{Kahneman \& Tversky 1979}:
  \[
    v(x) =
    \begin{cases}
      x^\alpha & x\ge 0 \\
      -\kappa\,|x|^\beta & x < 0
    \end{cases}
    \qquad (\alpha=\beta=0.88,\;\kappa=2.25).
  \]
  \begin{itemize}
    \item Sign of $r_i\tilde\delta_k$: \emph{positive} if perturbation moves \emph{with} persona $i$, \emph{negative} otherwise.
    \item $\kappa = 2.25$ → losses are weighted $\approx$2.25$\times$ more than gains → \textbf{loss-averse alignment}.
    \item Avoids committing to a single $\delta_\text{target}$, which the multi-persona setup is designed to avoid.
  \end{itemize}
\end{frame}

\begin{frame}{Stage 4b — Collective objective + MPPI weights}
  Free-energy collective utility:
  \[
    U_\text{total}(\epsilon_k) \;=\; \frac{1}{N}\sum_{i=1}^N U_i(\epsilon_k)
    \;-\; \alpha_\text{KL}\,\frac{\epsilon_k^2}{2\sigma^2},
    \qquad \alpha_\text{KL}=0.05.
  \]
  Importance-weighted MPPI update with $K=128$ samples:
  \[
    w_k = \frac{\exp(U_\text{total}(\epsilon_k)/\beta)}{\sum_{k'} \exp(U_\text{total}(\epsilon_{k'})/\beta)},
    \qquad
    \delta^\ast = \sum_{k=1}^{K} w_k\,\epsilon_k,\qquad
    \epsilon_k\sim\mathcal{N}(0,\sigma^2).
  \]
  \begin{block}{Key design choice}
    $\beta = T_\text{dec} = 0.5$: aligns MPPI smoothing scale with the final sigmoid decision temperature.
    Effective sample size $K_\text{eff} = 1/\sum_k w_k^2 \approx 15\text{--}40$ out of 128 → \emph{moderate concentration} (balances exploration with exploitation).
  \end{block}
\end{frame}

\begin{frame}{Stage 4c — Positional debiasing \& final prediction}
  \begin{itemize}
    \item \textbf{Recency / position bias} is a documented LLM failure mode. \cit{Liu et al. 2024 ``Lost in the Middle''}
    \item Each scenario is evaluated \textbf{twice}:
      \begin{itemize}
        \item original ordering, and
        \item with \textbf{both} choice labels (\texttt{A}$\leftrightarrow$\texttt{B}) and group labels (Group~A $\leftrightarrow$ Group~B) swapped \emph{in the native script}.
      \end{itemize}
    \item Debiased gap:
      \[
        \delta_i = \tfrac{1}{2}\bigl(\delta_i^{(\text{orig})} - \delta_i^{(\text{swap})}\bigr).
      \]
    \item Final prediction:
      \[
        p_\text{spare} = \sigma\!\Bigl((\bar\delta + \delta^\ast)\,/\,T_\text{dec}\Bigr),
        \quad T_\text{dec}=0.5.
      \]
  \end{itemize}
\end{frame}

\section{Experimental Setup}
% =============================================================================

\begin{frame}{Benchmark, languages, preprocessing}
  \begin{itemize}
    \item \textbf{MultiTP} \cit{Jin et al. 2025}: trolley vignettes, native languages, 6 moral dimensions:
      \emph{Species, Gender, Age, Fitness, Social Value, Utilitarianism}.
    \item \textbf{15 countries, 12 languages}: en, zh, ja, ko, de, fr, pt, ar, vi, hi, ru, es.
    \item \textbf{Native-language evaluation}: framing, characters, lane labels, questions all in target language; only \texttt{A}/\texttt{B} decision tokens remain English (verified single-token in every model vocabulary).
    \item \textbf{Preprocessing}: dedup → quality filter (drop equal-size Utilitarianism) → category cap (max 80) → up-sample (min 36 / dim) → 310--500 scenarios per country.
    \item Adaptive $\tau^{(c)}$ calibrated on 50 unlabelled scenarios (variance only, no AMCE).
  \end{itemize}
\end{frame}

\begin{frame}{Models, decoding, baselines, metrics}
  \textbf{Six open-weight LLMs} (single H100 80GB, Unsloth, greedy decoding):
  \begin{itemize}\small
    \item Qwen2.5-72B-Instruct (Int8) \cit{Qwen team 2025}; Qwen2.5-32B / 7B (4-bit)
    \item Llama-3.1-70B-Instruct (FP8) \cit{Grattafiori et al. 2024}; Llama-3.1-8B (4-bit)
    \item Mistral-Large-2407 (4-bit) \cit{Mistral AI 2024}
  \end{itemize}
  \textbf{Baselines on Llama-3.1-70B}: Vanilla; Country-Tailored (B1); WVS-Profile (B2); PRISM-style (B3) \cit{Kirk et al. 2024}; Activation Steering \cit{Arditi et al. 2025}.

  \textbf{Metrics}:
  \begin{itemize}\small
    \item \textbf{JSD} (primary, $\downarrow$): Jensen--Shannon distance between model and human AMCE vectors.
    \item \textbf{Pearson $r$} ($\uparrow$): correlation across 6D AMCE vector.
    \item \textbf{MAE} ($\downarrow$): mean absolute AMCE error.
    \item Bootstrap 95\% CIs (1000 resamples) → noise floor $\pm 0.004$ JSD.
  \end{itemize}
  \begin{alertblock}{Honest disclosure}
    All results are \textbf{single-seed} (seed 42); $K=128$ MPPI draws per scenario. Across-seed variance not characterised.
  \end{alertblock}
\end{frame}

\begin{frame}{Reading the four regimes}
  \footnotesize
  \begin{block}{\small Regime 1 — Strong 70B-class backbones (main claim)}
    Qwen2.5-72B and Llama-3.1-70B: vanilla JSD high enough to leave real headroom; logits well-spread; persona disagreement informative. \textbf{Mean JSD drops from $\sim$0.09 → $\sim$0.05} ($\Delta\!\approx\!0.032$ absolute, $-37.6\%$ / $-34.7\%$). Pearson $r$ moves into a meaningfully positive range. \textbf{This is where the main empirical claim lives.}
  \end{block}
  \begin{block}{\small Regime 2 — Small 4-bit quantised models (cautious read)}
    Qwen2.5-7B, Llama-3.1-8B: low vanilla JSD partly explained by 4-bit logit flattening toward chance. SWA-MPPI yields directional improvement, but \% gain is amplified by a small denominator.
  \end{block}
  \begin{block}{\small Regime 3 — Mistral-Large (informative negative)}
    JSD improves only marginally; Pearson $r$ \emph{decreases for 11/15 countries}. Hypothesis: on already-aligned logits, MPPI compresses the AMCE vector toward the population mean, improving distance but degrading rank. \textbf{Reported, not buried.}
  \end{block}
  \begin{block}{\small Regime 4 — Qwen2.5-32B (diagnosed failure)}
    Logit concentration at decision token: fixed $\sigma=0.3$ perturbations land outside effective logit range → importance weights $\to$ uniform → MPPI degenerates to random noise. Logit entropy $H = 0.521$ nats vs.\ reference $H_\text{ref}=2.5$. \textbf{Model-level diagnostic, not silent failure.}
  \end{block}
\end{frame}

\section{Mechanism \& Robustness}
% =============================================================================

\begin{frame}{The ``Anglosphere paradox''}
  \begin{block}{Observation}
    The largest JSD improvements cluster in the \textbf{three English-speaking countries} (AU, GB, US), reaching \textbf{+71--75\% on Qwen2.5-72B}. These are exactly the populations one would naively expect to be \emph{best} served by an English-trained model.
  \end{block}
  \begin{itemize}
    \item Vanilla JSDs for AU/GB/US are the \emph{highest} of any country evaluated.
    \item Whatever the strong 70B Qwen/Llama models have learned about ``the average moral subject'' lies further from the Moral Machine AMCEs of these populations than from Germany, China, etc.
    \item \textbf{We deliberately do not attribute this post-hoc to RLHF}---we have no intervention isolating RLHF as the cause.
    \item Reported \emph{as an open question deserving a dedicated follow-up}, not as an explained insight.
  \end{itemize}
\end{frame}

\begin{frame}{What is doing the work? (Mechanism analysis)}
  \footnotesize
  Two families of stripped-down baselines that share SWA-MPPI's infrastructure (multi-agent forward pass, two-pass debiasing) and differ \emph{only} in the decoding rule:

  \textbf{Family A — Consensus-shift baselines} (deterministic shifts on $\bar\delta$):
  \begin{itemize}
    \item Best variant (\textsc{CS-Clamp}): magnitude clamping closes \emph{most} of the gap.
    \item SWA-MPPI still wins by an absolute $\sim$\textbf{0.003 JSD} → residual = \textbf{Prospect-Theory asymmetry} on top of bounded perturbations.
  \end{itemize}

  \textbf{Family B — Binary-ARGS} (reward-guided decoding kernel):
  \begin{itemize}
    \item \textsc{ARGS-Unif}: uniform weighting → matches PersonaConsensus.
    \item \textsc{ARGS-WVS} (WVS-salience weighting, no PT): \emph{worse than uniform in 12/15 countries}. → \textbf{WVS weighting alone does not help}.
    \item \textsc{ARGS-PT} (Prospect-Theory utility kernel): JSD = 0.0485, $r$ = 0.514, best/tied in \textbf{13/15 countries} → recovers SWA-MPPI.
  \end{itemize}
  \begin{block}{\small Conclusion}
    The active ingredient is the \textbf{Prospect-Theory gain/loss asymmetry operating on bounded, KL-regularised perturbations}---exactly what the SWA-MPPI free-energy objective prescribes.
  \end{block}
\end{frame}

\section{Discussion \& Limitations}
% =============================================================================

\begin{frame}{Discussion: failure patterns are model-specific}
  \footnotesize
  \begin{itemize}
    \item \textbf{Vietnam on Llama-3.1-70B}: vanilla $r=-0.519$, SWA-MPPI $r=-0.135$. Headline JSD improvement is +65.8\%, but 6-D rank ordering is \emph{still wrong}. Most plausible reading: correction compresses Vietnam's AMCE vector toward population mean, which happens to lie closer to Vietnamese targets than vanilla. \textbf{Not direction-aware alignment in any strong sense.}
    \item \textbf{Mistral-Large}: JSD +5.8\% but Pearson $r$ \emph{decreases for 11/15 countries} (mean $\Delta r = -0.108$). Same JSD/$r$ dissociation phenomenon at panel level.
    \item \textbf{Qwen2.5-32B}: degrades in 7/15 countries—logit concentration confirmed.
  \end{itemize}
  \begin{block}{Practical implication}
    SWA-MPPI's effectiveness depends on the base model's logit calibration quality. A \textbf{pre-flight diagnostic} (e.g., logit entropy) could predict when to activate the framework.
  \end{block}
\end{frame}

\begin{frame}{Limitations (in order of how much they should temper confidence)}
  \footnotesize
  \begin{enumerate}
    \item \textbf{Hyperparameter validation is the weakest link.} Five hyperparameters tuned on 200 GPT-4-generated dilemmas with author proxy labels. The 50-scenario MultiTP transfer check is too small. Temperature/dataset sweeps in Appendix are the strongest empirical defence.
    \item \textbf{Single seed.} One run per model with one MPPI random draw per scenario; across-seed variance not characterised.
    \item \textbf{Pearson $r$ in 6D is noisy.} Individual $r$ deltas are directional, not statistically separable. JSD is the metric our claims rest on.
    \item \textbf{Mistral-Large rank-order regression.} JSD improves but $r$ decreases on 11/15 countries—no clean fix.
    \item \textbf{PT parameters fixed a priori.} $\alpha=\beta=0.88$, $\kappa=2.25$ from Western experiments; not validated per-culture.
    \item \textbf{Utilitarian persona dominance.} The single most important component—may overwhelm country-specific personas on non-numerosity dimensions (open question).
    \item \textbf{No per-dimension AMCE breakdown.} Aggregating over 6 dimensions hides which dimensions improve / hurt.
    \item \textbf{No human evaluation}; \textbf{WVS-to-trolley linkage indirect}; \textbf{coverage}: 15 of 100+ MultiTP countries.
  \end{enumerate}
\end{frame}

\begin{frame}{Broader impact \& safeguards}
  \begin{itemize}
    \item \textbf{Risk}: aligning to harmful majority preferences (e.g., discriminatory norms within populations); reifying stereotypes embedded in WVS averages.
    \item Four explicit safeguards proposed:
      \begin{enumerate}\small
        \item \textbf{Governance oversight}: SWA-MPPI must \emph{not} override safety-critical guardrails; deployment requires domain experts familiar with target culture.
        \item \textbf{Periodic persona updates}: refresh as newer WVS waves become available (current Wave 7: 2017--2022) to prevent temporal staleness.
        \item \textbf{Robustness checks}: audit per-dimension AMCE shifts to detect persona-driven entrenchment of harmful norms.
        \item \textbf{Explicit ethical bounds in the utility}: clip $\delta^\ast$ on sensitive dimensions (Gender, Social Value) by hard cap or steep penalty in $U_\text{total}$ → formalising deployment-time audits as decoding-time constraints.
      \end{enumerate}
  \end{itemize}
\end{frame}

\section{Conclusion}
% =============================================================================

\begin{frame}{Contributions (summary)}
  \begin{enumerate}
    \item \textbf{First MPPI in logit space} for cross-cultural moral alignment of LLMs—training-free, black-box, country-conditioned, label-free, weight-preserving.
    \item \textbf{Persona-disagreement gating} of a Prospect-Theory-weighted MPPI correction: intervention only fires when the decision is culturally contested (self-limiting).
    \item \textbf{Empirical evidence on six open-weight models} (7B--126B); strongest, cleanest gains on 70B-class backbones (\textbf{$-$37.6\% / $-$34.7\%} mean JSD).
    \item \textbf{Mechanism analysis} via consensus-shift + Binary-ARGS baselines: WVS weighting alone insufficient → \textbf{Prospect-Theory asymmetry on KL-regularised perturbations} is the load-bearing ingredient.
    \item \textbf{Robustness suite} (11 tests, all pass): rules out preprocessing, temperature, $\tau$ leakage as confounders.
    \item \textbf{Honest reporting of failure modes}: Mistral-Large rank-order regression, Qwen2.5-32B logit concentration, Anglosphere pattern as open question.
  \end{enumerate}
\end{frame}

\begin{frame}{Future directions}
  \begin{itemize}
    \item \textbf{Adaptive $\sigma$} from per-model logit entropy → broader compatibility (handles Qwen2.5-32B failure mode).
    \item \textbf{Beyond binary forced-choice}: multi-option / open-ended moral reasoning → lift persona-disagreement signal to a distribution over rationales.
    \item \textbf{Per-country / learned PT parameters} (relax fixed $\kappa=2.25$).
    \item \textbf{Per-dimension mediation analysis} to validate the WVS-to-trolley mapping.
    \item \textbf{Newer WVS waves} → longitudinal drift assessment.
    \item \textbf{Disentangle language vs.\ culture} via multilingual evaluation within single countries.
    \item \textbf{Decoding-time ethical constraints}: formalise the safeguards inside $U_\text{total}$.
  \end{itemize}
\end{frame}
